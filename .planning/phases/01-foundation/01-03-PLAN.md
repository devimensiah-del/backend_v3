### Task 1: Create logger wrapper
**Type:** create
**Files:** `pkg/logging/logger.go`
**Action:**
```go
package logging

import (
    "os"
    "github.com/rs/zerolog"
)

type Logger struct {
    zerolog.Logger
}

func NewLogger(env string) *Logger {
    level := zerolog.InfoLevel
    if env == "development" {
        level = zerolog.DebugLevel
    }

    logger := zerolog.New(os.Stdout).
        With().
        Timestamp().
        Caller().
        Logger().
        Level(level)

    return &Logger{logger}
}

// Contextual field methods
func (l *Logger) WithRequestID(requestID string) *Logger {
    return &Logger{l.With().Str("request_id", requestID).Logger()}
}

func (l *Logger) WithSubmissionID(id string) *Logger {
    return &Logger{l.With().Str("submission_id", id).Logger()}
}

func (l *Logger) WithUserID(id string) *Logger {
    return &Logger{l.With().Str("user_id", id).Logger()}
}

func (l *Logger) WithComponent(name string) *Logger {
    return &Logger{l.With().Str("component", name).Logger()}
}

func (l *Logger) WithEnrichmentID(id string) *Logger {
    return &Logger{l.With().Str("enrichment_id", id).Logger()}
}

func (l *Logger) WithAnalysisID(id string) *Logger {
    return &Logger{l.With().Str("analysis_id", id).Logger()}
}

func (l *Logger) WithJobID(id string) *Logger {
    return &Logger{l.With().Str("job_id", id).Logger()}
}

func (l *Logger) WithDuration(ms int64) *Logger {
    return &Logger{l.With().Int64("duration_ms", ms).Logger()}
}
```

### Task 2: Create middleware helper
**Type:** create
**Files:** `pkg/logging/middleware.go`
**Action:**
Create Gin middleware that:
- Generates request_id (UUID)
- Extracts user_id from JWT claims if present
- Attaches logger to context
- Logs request start/end with duration

```go
func GinMiddleware(base *Logger) gin.HandlerFunc {
    return func(c *gin.Context) {
        requestID := uuid.New().String()
        start := time.Now()

        logger := base.WithRequestID(requestID)

        // Extract user_id if authenticated
        if userID, exists := c.Get("user_id"); exists {
            logger = logger.WithUserID(userID.(string))
        }

        c.Set("logger", logger)
        c.Header("X-Request-ID", requestID)

        c.Next()

        duration := time.Since(start).Milliseconds()
        logger.Info().
            Str("method", c.Request.Method).
            Str("path", c.Request.URL.Path).
            Int("status", c.Writer.Status()).
            Int64("duration_ms", duration).
            Msg("request completed")
    }
}
```

**Verify:** `go build ./pkg/logging/`

### Task 3: Create context helper
**Type:** create
**Files:** `pkg/logging/context.go`
**Action:**
```go
package logging

import (
    "context"
    "github.com/gin-gonic/gin"
)

type ctxKey struct{}

func FromContext(ctx context.Context) *Logger {
    if l, ok := ctx.Value(ctxKey{}).(*Logger); ok {
        return l
    }
    // Return default logger if none in context
    return NewLogger("production")
}

func ToContext(ctx context.Context, logger *Logger) context.Context {
    return context.WithValue(ctx, ctxKey{}, logger)
}

func FromGin(c *gin.Context) *Logger {
    if l, exists := c.Get("logger"); exists {
        return l.(*Logger)
    }
    return NewLogger("production")
}
```

**Verify:** `go build ./pkg/logging/`
