## Tasks

### Task 1: Create error response helper
**Type:** modify
**Files:** `api/helpers.go`
**Action:**
Add new helper function:
```go
import (
    apperrors "backend_v3/pkg/errors"
)

func RespondError(c *gin.Context, err error) {
    // Get language preference from Accept-Language header
    lang := c.GetHeader("Accept-Language")
    if lang == "" {
        lang = "pt-BR" // Default to Portuguese for Brazilian market
    }

    // Check if it's an AppError
    if appErr, ok := apperrors.AsAppError(err); ok {
        c.JSON(appErr.HTTPStatus, gin.H{
            "error": gin.H{
                "code":    appErr.Code,
                "message": appErr.LocalizedMessage(lang),
            },
        })
        return
    }

    // Fallback for non-AppError
    c.JSON(500, gin.H{
        "error": gin.H{
            "code":    "ERR_INTERNAL",
            "message": apperrors.ErrInternal.LocalizedMessage(lang),
        },
    })
}

// RespondValidationError for field-specific errors
func RespondValidationError(c *gin.Context, field, message string) {
    lang := c.GetHeader("Accept-Language")
    if lang == "" {
        lang = "pt-BR"
    }

    c.JSON(400, gin.H{
        "error": gin.H{
            "code":    "ERR_VALIDATION",
            "message": apperrors.ErrValidation.LocalizedMessage(lang),
            "details": gin.H{
                "field":   field,
                "message": message,
            },
        },
    })
}
```

**Verify:** `go build ./api/...`
